cmake_minimum_required(VERSION 3.16)
project(aps_ordenacao LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(aps src/main.c src/dados.c src/ordenacao.c src/tempo.c src/estruturas.c src/experimentos.c src/escala.c)
target_include_directories(aps PRIVATE include)
if(MSVC)
    target_compile_options(aps PRIVATE /W4)
    target_compile_definitions(aps PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(aps PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(aps PRIVATE m)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(aps_testes tests/test_core.c src/dados.c src/ordenacao.c src/tempo.c src/estruturas.c)
    target_include_directories(aps_testes PRIVATE include)
    add_test(NAME nucleo COMMAND aps_testes)
endif()
